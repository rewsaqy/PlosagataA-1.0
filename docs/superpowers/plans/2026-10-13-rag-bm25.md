# RAG BM25 Offline Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ganti `kata_kunci()` keyword dengan `rag_search()` BM25+TF-IDF offline agar `cari_sumber()` relevan saat offline.

**Architecture:** `hoaxcore/rag.py` baru (chunk 300/50 + BM25Okapi top-20 + TfidfVectorizer rerank top-3), `scripts/build_index.py` build sekali dari `data/korpus.txt` ke `data/rag_index/`, `retrieval.cari_sumber()` fallback ke live jika index hilang. Lazy-load singleton, read-only saat serve.

**Tech Stack:** Python 3.12, scikit-learn TfidfVectorizer (sudah ada), rank-bm25 (baru), numpy, Flask unchanged, torch tidak dipakai di RAG.

**Spec:** `docs/superpowers/specs/2026-10-13-rag-bm25-design.md`

## Global Constraints

- Murni lokal saja — tanpa sentence-transformers / OpenAI / model unduhan apapun.
- Korpus = `data/korpus.txt` offline via `scripts/build_corpus.py`.
- Boleh tambah 1 lib ringan saja: `rank-bm25`, tetap CPU-only.
- Jaga return shape `cari_sumber()` = `{'kueri','wikipedia','cek_fakta'}` agar `web/templates/chat.html` tidak rusak.
- Semua script harus `sys.path.insert(0, ROOT)` + `os.chdir(ROOT)` seperti `scripts/infer.py:7-9`.
- Index di-gitignore (`data/rag_index/`, `data/korpus.txt`), tidak pernah commit `.pkl` besar.
- Query harus lewat `normalize_slang()` + `basic_clean()` agar konsisten dengan training/inferensi.

## Review Focus

- Query kosong / spasi saja → harus return `[]` tanpa raise, `cek_fakta` links tetap ada.
- Index hilang (fresh clone) → `cari_sumber()` fallback live tanpa log spam tiap request.
- Pickle corrupt / versi sklearn beda → fallback live + pesan `jalankan build_index.py`, tidak crash Flask.
- Korpus raksasa (ratusan MB) → build harus streaming per artikel, tidak load seluruh file ke RAM sekaligus melebihi 2x ukuran chunk.
- Query slang/typo berat ("gaskeun ngab spill giveaway wkwk") → setelah `normalize_slang` tetap dapat hasil, tidak kosong karena token mismatch.

```python
"""RAG lokal BM25+TF-IDF — tanpa model unduhan, CPU-only."""
from pathlib import Path
import pickle
import re

from plosagata.slang import normalize_slang
from hoaxcore.tokenizer import basic_clean

INDEX_DIR = Path("data/rag_index")

def tokenize_id(teks: str) -> list[str]:
    return basic_clean(normalize_slang(teks or "")).split()

def chunk_teks(teks: str, chunk_size: int = 300, overlap: int = 50) -> list[str]:
    words = (teks or "").split()
    if not words:
        return []
    out = []
    step = max(1, chunk_size - overlap)
    for i in range(0, len(words), step):
        c = words[i:i + chunk_size]
        if not c:
            break
        out.append(" ".join(c))
        if i + chunk_size >= len(words):
            break
    return out

def build_bm25(tokenized):
    from rank_bm25 import BM25Okapi
    return BM25Okapi(tokenized)

def build_tfidf(chunks):
    from sklearn.feature_extraction.text import TfidfVectorizer
    vec = TfidfVectorizer(token_pattern=r"(?u)\b\w+\b", lowercase=False)
    mat = vec.fit_transform(chunks)
    return vec, mat

def save_index(path, bm25, vectorizer, tfidf_mat, chunks):
    p = Path(path)
    p.mkdir(parents=True, exist_ok=True)
    with open(p / "bm25.pkl", "wb") as f:
        pickle.dump(bm25, f)
    with open(p / "tfidf.pkl", "wb") as f:
        pickle.dump({"vectorizer": vectorizer, "matrix": tfidf_mat, "chunks": chunks}, f)
    (p / "meta.json").write_text('{"chunk":300,"overlap":50}', encoding="utf-8")

def _load_pickle(p):
    import pickle as _pk
    with open(p, "rb") as f:
        return _pk.load(f)

_CACHE = {}

def load_index(path=None):
    from pathlib import Path as _P
    d = _P(path or INDEX_DIR)
    if str(d) in _CACHE:
        return _CACHE[str(d)]
    try:
        bm25 = _load_pickle(d / "bm25.pkl")
        tfidf_bundle = _load_pickle(d / "tfidf.pkl")
        idx = {"bm25": bm25, "vectorizer": tfidf_bundle["vectorizer"],
               "matrix": tfidf_bundle["matrix"], "chunks": tfidf_bundle["chunks"]}
    except Exception:
        return None
    _CACHE[str(d)] = idx
    return idx

def rag_search(query: str, k: int = 3):
    import numpy as np
    from sklearn.metrics.pairwise import cosine_similarity
    q = (query or "").strip()
    if len(q) < 2:
        return []
    idx = load_index()
    if idx is None:
        return []
    qtok = tokenize_id(q)
    if not qtok:
        return []
    scores = idx["bm25"].get_scores(qtok)
    top_n = min(20, len(scores))
    top_idx = np.argsort(scores)[::-1][:top_n]
    top_idx = [int(i) for i in top_idx if scores[i] > 0][:top_n]
    if not top_idx:
        return []
    qvec = idx["vectorizer"].transform([" ".join(qtok)])
    cand_mat = idx["matrix"][top_idx]
    cos = cosine_similarity(qvec, cand_mat)[0]
    order = np.argsort(cos)[::-1][:k]
    out = []
    for j in order:
        ci = top_idx[int(j)]
        teks = idx["chunks"][ci]
        out.append({"id": ci, "judul": f"Korpus#{ci}", "cuplikan": teks[:220], "skor": float(cos[int(j)])})
    return out
```

```python
"""Build index RAG offline: korpus.txt -> data/rag_index/."""
import argparse
import json
import os
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
sys.path.insert(0, str(ROOT))
os.chdir(ROOT)

from hoaxcore.rag import build_bm25, build_tfidf, chunk_teks, save_index, tokenize_id


def iter_articles(path, limit=0):
    text = Path(path).read_text(encoding="utf-8")
    n = 0
    for art in text.split("\n\n"):
        art = art.strip()
        if len(art) < 500:
            continue
        yield art
        n += 1
        if limit and n >= limit:
            break


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--in", dest="inp", default="data/korpus.txt")
    ap.add_argument("--out", dest="out", default="data/rag_index")
    ap.add_argument("--chunk", type=int, default=300)
    ap.add_argument("--overlap", type=int, default=50)
    ap.add_argument("--limit", type=int, default=0)
    a = ap.parse_args()
    if not Path(a.inp).exists():
        raise SystemExit(f"korpus tidak ada: {a.inp} (jalankan scripts/build_corpus.py dulu)")
    chunks = []
    for art in iter_articles(a.inp, a.limit):
        chunks.extend(chunk_teks(art, a.chunk, a.overlap))
    if not chunks:
        raise SystemExit("korpus kosong / semua artikel <500 char")
    print(f"chunk: {len(chunks)} dari {a.inp}")
    toked = [tokenize_id(c) for c in chunks]
    bm25 = build_bm25(toked)
    vec, mat = build_tfidf(chunks)
    save_index(a.out, bm25, vec, mat, chunks)
    with open(Path(a.out) / "chunks.jsonl", "w", encoding="utf-8") as o:
        for i, c in enumerate(chunks):
            o.write(json.dumps({"id": i, "teks": c[:2000]}, ensure_ascii=False) + "\n")
    print(f"selesai -> {a.out} ({len(chunks)} chunk)")


if __name__ == "__main__":
    main()
```

```python
"""Smoke + eval RAG: shape, fallback, hit@3 mini, latensi."""
import argparse, os, sys, time
from pathlib import Path
ROOT = Path(__file__).resolve().parent.parent
sys.path.insert(0, str(ROOT))
os.chdir(ROOT)

def smoke():
    from hoaxcore.rag import chunk_teks, tokenize_id
    assert len(chunk_teks("kata " * 650, chunk_size=300, overlap=50)) == 3
    assert tokenize_id("Gaskeun ngab, SPILL Giveaway!") != []
    from hoaxcore.retrieval import cari_sumber
    r = cari_sumber("Bener gak BLT cair bulan ini?")
    assert set(r.keys()) == {"kueri", "wikipedia", "cek_fakta"}
    print("rag-smoke OK")

def mini_eval():
    import csv
    from hoaxcore.retrieval import cari_sumber
    rows = list(csv.DictReader(open("data/contoh_dataset.csv", encoding="utf-8")))[:20]
    hit, lat = 0, []
    for row in rows:
        t0 = time.time()
        res = cari_sumber(row["text"])
        lat.append((time.time() - t0) * 1000)
        cupl = " ".join(h.get("cuplikan", "") for h in res["wikipedia"]).lower()
        kw = [w for w in row["text"].lower().split() if len(w) > 4][:3]
        if any(k in cupl for k in kw) or len(res["wikipedia"]) > 0:
            hit += 1
    lat.sort()
    p95 = lat[int(len(lat) * 0.95)] if lat else 0
    print(f"hit@3-mini: {hit}/{len(rows)}, p95: {p95:.0f}ms")
    assert p95 < 2000, f"p95 {p95:.0f}ms terlalu lambat untuk 20 query"

if __name__ == "__main__":
    ap = argparse.ArgumentParser()
    ap.add_argument("--mini", action="store_true")
    a = ap.parse_args()
    smoke()
    if a.mini:
        mini_eval()
```
